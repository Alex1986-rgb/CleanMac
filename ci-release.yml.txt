# (нужен gh-токен со scope 'workflow' или коммит через веб-интерфейс GitHub).

name: Build & Release

on:
  push:
    tags: ["v*"]
  workflow_dispatch:

permissions:
  contents: write

jobs:
  build-dmg:
    runs-on: macos-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"

      - name: Install build deps
        run: pip install --upgrade pip pyinstaller pillow

      - name: Build universal .app + .dmg
        run: |
          pyinstaller --windowed --name CleanMac --icon CleanMac.icns \
            --target-arch universal2 \
            --osx-bundle-identifier com.macbook.cleanmac --noconfirm CleanMac.py
          STAGE="$(mktemp -d)/dmg"; mkdir -p "$STAGE"
          cp -R dist/CleanMac.app "$STAGE/"
          ln -s /Applications "$STAGE/Applications"
          hdiutil create -volname CleanMac -srcfolder "$STAGE" -ov -format UDZO CleanMac.dmg
          shasum -a 256 CleanMac.dmg

      - name: Run unit tests
        run: python -m unittest -v tests.test_cleanmac

      - name: Attach DMG to release
        if: startsWith(github.ref, 'refs/tags/')
        uses: softprops/action-gh-release@v2
        with:
          files: CleanMac.dmg
