# Чтобы включить CI: переместите этот файл в .github/workflows/release.yml
# (нужен gh-токен со scope 'workflow' или коммит через веб-интерфейс GitHub).
#
# Собирает по тегу vX.Y.Z: macOS DMG (CleanMac) + Windows .exe + Linux (KRYLAN Desktop),
# прогоняет тесты и прикрепляет всё к релизу.

name: Build & Release (all platforms)

on:
  push:
    tags: ["v*"]
  workflow_dispatch:

permissions:
  contents: write

jobs:
  # ---------- macOS: нативный CleanMac (universal DMG) ----------
  macos:
    runs-on: macos-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with: { python-version: "3.12" }
      - run: pip install --upgrade pip pyinstaller pillow
      - run: python -m unittest -v tests.test_cleanmac
      - name: Build CleanMac DMG
        run: |
          pyinstaller --windowed --name CleanMac --icon CleanMac.icns \
            --target-arch universal2 \
            --osx-bundle-identifier com.macbook.cleanmac --noconfirm CleanMac.py
          STAGE="$(mktemp -d)/dmg"; mkdir -p "$STAGE"
          cp -R dist/CleanMac.app "$STAGE/"; ln -s /Applications "$STAGE/Applications"
          hdiutil create -volname CleanMac -srcfolder "$STAGE" -ov -format UDZO CleanMac.dmg
      - uses: softprops/action-gh-release@v2
        if: startsWith(github.ref, 'refs/tags/')
        with: { files: CleanMac.dmg }

  # ---------- Windows: KRYLAN Desktop (.exe) ----------
  windows:
    runs-on: windows-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with: { python-version: "3.12" }
      - run: pip install --upgrade pip pyinstaller -r krylan-desktop/requirements.txt
      - name: Build KRYLAN.exe
        run: pyinstaller --onefile --windowed --name KRYLAN --icon krylan-desktop/krylan.ico krylan-desktop/krylan.py
      - uses: softprops/action-gh-release@v2
        if: startsWith(github.ref, 'refs/tags/')
        with: { files: dist/KRYLAN.exe }

  # ---------- Linux: KRYLAN Desktop (бинарь) ----------
  linux:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with: { python-version: "3.12" }
      - run: |
          sudo apt-get update && sudo apt-get install -y python3-tk
          pip install --upgrade pip pyinstaller -r krylan-desktop/requirements.txt
      - name: Build KRYLAN (Linux)
        run: pyinstaller --onefile --name KRYLAN-linux krylan-desktop/krylan.py
      - uses: softprops/action-gh-release@v2
        if: startsWith(github.ref, 'refs/tags/')
        with: { files: dist/KRYLAN-linux }
